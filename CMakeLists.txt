cmake_minimum_required(VERSION 3.24)
project(RainstarPaint VERSION 0.1.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
endif()
option(RAINSTAR_SYSTEM_SDL "Use installed SDL3 for development" ON)
include(FetchContent)
if(RAINSTAR_SYSTEM_SDL)
  find_package(SDL3 CONFIG REQUIRED)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.4.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(SDL3)
endif()
option(RAINSTAR_BUNDLED_TIFF "Build the pinned TIFF library with permissive codecs" OFF)
if(RAINSTAR_BUNDLED_TIFF)
  foreach(option tiff-tools tiff-tests tiff-contrib tiff-docs tiff-cxx tiff-install jbig lerc libdeflate lzma zstd webp)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(tiff-static ON CACHE BOOL "" FORCE)
  FetchContent_Declare(tiff GIT_REPOSITORY https://gitlab.com/libtiff/libtiff.git GIT_TAG v4.7.1 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(tiff)
  set(RAINSTAR_TIFF_TARGET TIFF::tiff)
else()
  find_package(TIFF REQUIRED)
  set(RAINSTAR_TIFF_TARGET TIFF::TIFF)
endif()
find_package(PkgConfig QUIET)
find_package(WebP CONFIG QUIET)
if(TARGET WebP::webp)
  set(RAINSTAR_WEBP_TARGET WebP::webp)
else()
  find_path(WEBP_INCLUDE_DIR webp/encode.h REQUIRED)
  find_library(WEBP_LIBRARY NAMES webp REQUIRED)
  set(RAINSTAR_WEBP_TARGET ${WEBP_LIBRARY})
endif()
add_library(paint_core src/image.cpp src/conv.cpp src/raster.cpp src/document.cpp src/codecs.cpp src/warp.cpp)
target_include_directories(paint_core PUBLIC src PRIVATE vendor/stb vendor/gif-h ${WEBP_INCLUDE_DIR})
target_link_libraries(paint_core PUBLIC ${RAINSTAR_TIFF_TARGET} ${RAINSTAR_WEBP_TARGET})
add_library(imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp vendor/imgui/backends/imgui_impl_sdl3.cpp vendor/imgui/backends/imgui_impl_sdlrenderer3.cpp)
target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(imgui PUBLIC SDL3::SDL3)
add_executable(rainstar-paint MACOSX_BUNDLE src/main.cpp src/application.cpp src/renderer.cpp src/idle_render.cpp src/ribbon.cpp src/help.cpp src/platform.cpp src/desktop.cpp src/color_editor.cpp src/text.cpp src/embedded_font.cpp src/warp_session.cpp src/reshape.cpp src/rotation.cpp)
target_link_libraries(rainstar-paint PRIVATE paint_core imgui)
target_include_directories(rainstar-paint PRIVATE vendor/stb)
if(WIN32)
  target_sources(rainstar-paint PRIVATE packaging/windows.rc)
  target_include_directories(rainstar-paint PRIVATE assets)
  set_target_properties(rainstar-paint PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
set_target_properties(rainstar-paint PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Rainstar Paint" MACOSX_BUNDLE_GUI_IDENTIFIER "org.rainstar.paint" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
if(APPLE)
  target_sources(rainstar-paint PRIVATE assets/app-icon.icns)
  set_source_files_properties(assets/app-icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(rainstar-paint PROPERTIES MACOSX_BUNDLE_ICON_FILE app-icon.icns)
  enable_language(OBJCXX)
  target_sources(rainstar-paint PRIVATE src/platform_mac.mm)
  target_link_libraries(rainstar-paint PRIVATE "-framework Cocoa" "-framework UniformTypeIdentifiers")
endif()
if(MSVC)
  add_compile_options(/utf-8)
  target_compile_options(paint_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(rainstar-paint PRIVATE /utf-8)
else()
  target_compile_options(paint_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(rainstar-paint PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(paint-tests tests/core_tests.cpp)
target_link_libraries(paint-tests PRIVATE paint_core)
add_test(NAME paint-core COMMAND paint-tests)
install(TARGETS rainstar-paint BUNDLE DESTINATION . RUNTIME DESTINATION .)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)

add_executable(paint-ui-tests tests/ui_tests.cpp src/application.cpp src/renderer.cpp src/idle_render.cpp src/ribbon.cpp src/help.cpp src/platform.cpp src/desktop.cpp src/color_editor.cpp src/text.cpp src/embedded_font.cpp src/warp_session.cpp src/reshape.cpp src/rotation.cpp)
target_include_directories(paint-ui-tests PRIVATE vendor/stb)
target_link_libraries(paint-ui-tests PRIVATE paint_core imgui)
if(APPLE)
  target_sources(paint-ui-tests PRIVATE src/platform_mac.mm)
  target_link_libraries(paint-ui-tests PRIVATE "-framework Cocoa" "-framework UniformTypeIdentifiers")
endif()
add_test(NAME paint-interactions COMMAND paint-ui-tests)
add_executable(paint-warp-tests tests/warp_tests.cpp)
target_link_libraries(paint-warp-tests PRIVATE paint_core)
add_test(NAME paint-warp COMMAND paint-warp-tests)

if(WIN32)
  foreach(application rainstar-paint paint-ui-tests)
    target_sources(${application} PRIVATE src/print_windows.cpp src/desktop_windows.cpp)
    target_link_libraries(${application} PRIVATE comdlg32 gdi32 ole32 oleaut32 uuid)
  endforeach()
elseif(UNIX AND NOT APPLE)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  foreach(application rainstar-paint paint-ui-tests)
    target_sources(${application} PRIVATE src/print_linux.cpp src/desktop_linux.cpp)
    target_link_libraries(${application} PRIVATE PkgConfig::GTK3)
  endforeach()
endif()

option(RAINSTAR_BENCHMARKS "Build the standalone CONV timing and checksum harness" OFF)
if(RAINSTAR_BENCHMARKS)
  add_executable(paint-benchmark benchmarks/conv_benchmark.cpp)
  target_link_libraries(paint-benchmark PRIVATE paint_core)
endif()

option(RAINSTAR_COMPILER_REPORTS "Save CONV assembly and vectorization diagnostics" OFF)
if(RAINSTAR_COMPILER_REPORTS)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/compiler-reports")
  foreach(kernel conv warp)
    if(MSVC)
      set_property(SOURCE src/${kernel}.cpp APPEND PROPERTY COMPILE_OPTIONS
        /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/compiler-reports/${kernel}.asm")
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      set_property(SOURCE src/${kernel}.cpp APPEND PROPERTY COMPILE_OPTIONS
        -save-temps=obj "-fopt-info-vec-all=${CMAKE_CURRENT_BINARY_DIR}/compiler-reports/${kernel}.txt")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      set_property(SOURCE src/${kernel}.cpp APPEND PROPERTY COMPILE_OPTIONS
        -save-temps=obj "-Rpass=loop-vectorize|slp-vectorizer")
    endif()
  endforeach()
endif()
